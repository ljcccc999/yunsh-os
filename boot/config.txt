# YUNSH OS v1.0 - Raspberry Pi Boot Config
# EDID-driven output for mono monitors and binocular SBS display controllers.
# Black remains transparent on the target optical display.

# GPU memory
gpu_mem=256

# Let KMS select the display controller's native mode. A binocular controller
# can expose 3840x1080 (two 1920x1080 eyes); a normal monitor stays supported.
display_auto_detect=1
hdmi_drive=2
# Retain the firmware-side hotplug hint while KMS owns mode selection.
hdmi_force_hotplug=1

# Display settings
disable_overscan=1
framebuffer_depth=32
framebuffer_ignore_alpha=0

# Enable 64-bit kernel
arm_64bit=1

# Boot (Pi 5 auto-picks kernel8.img — no kernel= line needed)
# kernel= line removed: Pi 5 firmware searches kernel8.img by default

# Pi 5 firmware KMS overlay. The launcher can use the firmware framebuffer
# with Qt linuxfb if the VC4 firmware clock provider is unavailable.
dtoverlay=vc4-kms-v3d-pi5
max_framebuffers=2

# Serial console for debugging
enable_uart=1

# Audio
dtparam=audio=on

# USB
dtoverlay=dwc2,dr_mode=host
